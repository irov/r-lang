    "${R_LIBRARY_REPOSITORY_ROOT}/library/internal/math/source/binary_scale.c"
    "${R_LIBRARY_REPOSITORY_ROOT}/library/internal/math/source/complex.c"
    "${R_LIBRARY_REPOSITORY_ROOT}/library/internal/math/source/environment.c"
)

    "${R_LIBRARY_REPOSITORY_ROOT}/library/internal/math/include"
)

    target_compile_options(r_std_math PRIVATE -ffp-model=strict)
elseif(CMAKE_C_COMPILER_ID STREQUAL "GNU")
    target_compile_options(r_std_math PRIVATE -frounding-math -ftrapping-math)
endif()
target_link_libraries(r_std_math PUBLIC r_std_error)
if(NOT MSVC)
    target_link_libraries(r_std_math PUBLIC m)
endif()
